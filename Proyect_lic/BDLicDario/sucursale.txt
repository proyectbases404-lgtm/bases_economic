Table sucursales {
  id int [pk, increment]
  nombre varchar
  direccion text
  ciudad varchar
  estado boolean
}

Table usuarios {
  id int [pk, increment]
  sucursal_id int
  nombre varchar
  correo varchar [unique]
  contraseña varchar
  rol varchar
  estado boolean
}

Table categorias {
  id int [pk, increment]
  nombre varchar
  descripcion text
}

Table proveedores {
  id int [pk, increment]
  nombre varchar
  telefono varchar
  direccion text
  correo varchar
}

Table productos {
  id int [pk, increment]
  nombre varchar
  categoria_id int
  proveedor_id int
  precio_venta decimal
  codigo varchar
}

Table lotes {
  id int [pk, increment]
  sucursal_id int
  producto_id int
  numero_lote varchar
  fecha_vencimiento date
  cantidad_inicial int
  cantidad_actual int
  precio_compra decimal
  estado varchar
}

Table movimientos {
  id int [pk, increment]
  sucursal_id int
  lote_id int
  tipo varchar
  cantidad int
  fecha datetime
  referencia varchar
}

Table ubicaciones {
  id int [pk, increment]
  nombre varchar
}

Table stock_ubicacion {
  id int [pk, increment]
  sucursal_id int
  lote_id int
  ubicacion_id int
  cantidad int
}

Table ventas {
  id int [pk, increment]
  sucursal_id int
  usuario_id int
  fecha datetime
  total decimal
  estado varchar
}

Table detalle_venta {
  id int [pk, increment]
  venta_id int
  lote_id int
  cantidad int
  precio_unitario decimal
  subtotal decimal
}

Table tipos_pago {
  id int [pk, increment]
  nombre varchar
}

Table pagos {
  id int [pk, increment]
  venta_id int
  tipo_pago_id int
  monto decimal
  referencia varchar
  fecha datetime
}

Table compras {
  id int [pk, increment]
  sucursal_id int
  proveedor_id int
  usuario_id int
  fecha datetime
  total decimal
}

Table detalle_compra {
  id int [pk, increment]
  compra_id int
  producto_id int
  numero_lote varchar
  fecha_vencimiento date
  cantidad int
  precio_compra decimal
  subtotal decimal
}

/* =========================
   RELACIONES
   ========================= */

Ref: usuarios.sucursal_id > sucursales.id

Ref: productos.categoria_id > categorias.id
Ref: productos.proveedor_id > proveedores.id

Ref: lotes.producto_id > productos.id
Ref: lotes.sucursal_id > sucursales.id

Ref: movimientos.lote_id > lotes.id
Ref: movimientos.sucursal_id > sucursales.id

Ref: stock_ubicacion.lote_id > lotes.id
Ref: stock_ubicacion.ubicacion_id > ubicaciones.id
Ref: stock_ubicacion.sucursal_id > sucursales.id

Ref: ventas.usuario_id > usuarios.id
Ref: ventas.sucursal_id > sucursales.id

Ref: detalle_venta.venta_id > ventas.id
Ref: detalle_venta.lote_id > lotes.id

Ref: pagos.venta_id > ventas.id
Ref: pagos.tipo_pago_id > tipos_pago.id

Ref: compras.proveedor_id > proveedores.id
Ref: compras.usuario_id > usuarios.id
Ref: compras.sucursal_id > sucursales.id

Ref: detalle_compra.compra_id > compras.id
Ref: detalle_compra.producto_id > productos.id
